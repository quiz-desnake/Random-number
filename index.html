from flask import Flask, render_template_string, request import random app = Flask(__name__) HTML = """ Random Generator

🎲 Random Generator

Tasodifiy son generatsiya qiling

{% if number is not none %}
Sizning random soningiz:
{{ number }}
{% endif %} {% if error %}
❌ {{ error }}
{% endif %}
""" @app.route("/", methods=["GET", "POST"]) def home(): number = None error = "" min_number = "" max_number = "" if request.method == "POST": try: min_number = int(request.form["min_number"]) max_number = int(request.form["max_number"]) if min_number > max_number: error = "Minimum son Maximum sondan katta bo'lishi mumkin emas!" else: number = random.randint(min_number, max_number) except ValueError: error = "Iltimos, faqat son kiriting!" return render_template_string( HTML, number=number, error=error, min_number=min_number, max_number=max_number ) if __name__ == "__main__": app.run(debug=True)
